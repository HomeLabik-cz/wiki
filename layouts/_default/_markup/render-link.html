{{- $dest := .Destination -}} {{- $text := .Text -}} {{- $title := .Title -}} {{- $isTargetBlank := false -}} {{- /* External links open in new tab */ -}} {{- if or (strings.HasPrefix $dest "http://") (strings.HasPrefix $dest "https://") -}} {{- $isTargetBlank = true -}} {{- end -}} {{- /* Explicit _blank flag in title or URL suffix */ -}} {{- if or (eq $title "_blank") (strings.HasSuffix $dest " _blank") -}} {{- $isTargetBlank = true -}} {{- if eq $title "_blank" }}{{ $title = "" }}{{ end -}} {{- if strings.HasSuffix $dest " _blank" }}{{ $dest = strings.TrimSuffix " _blank" $dest }}{{ end -}} {{- end -}} {{ $text | safeHTML }}
